2 Call Center queue management


class CallCenter
    attributes: emergencyQueue, normalQueue

    method addCustomer(name, type)
        IF type == "emergency"
            enqueue name into emergencyQueue
        ELSE
            enqueue name into normalQueue
        print "Added name to type queue"

    method serveCustomer()
        IF emergencyQueue is not empty
            customer = dequeue from emergencyQueue
        ELSE IF normalQueue is not empty
            customer = dequeue from normalQueue
        ELSE
            print "No customers waiting"
            RETURN
        print "Serving: customer"
