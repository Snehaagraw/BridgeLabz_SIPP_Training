CLASS Item
    ATTRIBUTES: name, price
    CONSTRUCTOR(name, price)

CLASS ShoppingCart
    ATTRIBUTE: cartItems (List of Item)

    METHOD addItem(item)
        add item to cartItems

    GENERIC METHOD calculateTotalPrice(List<Item>)
        total = 0
        FOR each item in List
            total = total + item.price
        RETURN total

MAIN
    CREATE cart = new ShoppingCart
    cart.addItem(new Item("Shoes", 1500))
    cart.addItem(new Item("Bag", 800))
    cart.addItem(new Item("Watch", 2000))

    totalPrice = cart.calculateTotalPrice(cart.cartItems)
    PRINT "Total Price = " + totalPrice
