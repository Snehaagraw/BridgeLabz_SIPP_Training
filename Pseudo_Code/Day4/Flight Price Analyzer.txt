Function merge(left, right, comparator)
    result = []
    i = j = 0
    While i < left.size AND j < right.size
        If comparator.compare(left[i], right[j]) <= 0
            result.add(left[i])
            i++
        Else
            result.add(right[j])
            j++
    Add remaining elements
    return result
