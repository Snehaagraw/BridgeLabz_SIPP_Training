2.Reastaurant POS Billing System

Interface TaxCalculator
    Method calculateTax(amount: double) → double

Class FoodTax implements TaxCalculator
    Method calculateTax(amount)
        Return amount * 0.05

Class BeverageTax implements TaxCalculator
    Method calculateTax(amount)
        Return amount * 0.12

Class ImportedTax implements TaxCalculator
    Method calculateTax(amount)
        Return amount * 0.05 + customsDuty

// New Tax can be added like LuxuryTax
Class LuxuryTax implements TaxCalculator
    Method calculateTax(amount)
        Return amount * 0.18

Interface DiscountPolicy
    Method applyDiscount(total: double) → double

Class NoDiscount implements DiscountPolicy
    Method applyDiscount(total)
        Return total

Class FlatDiscount implements DiscountPolicy
    Method applyDiscount(total)
        Return total - 100

Class PercentageDiscount implements DiscountPolicy
    Method applyDiscount(total)
        Return total * 0.9   // 10% off

// New scheme easily pluggable
Class BuyOneGetOneDiscount implements DiscountPolicy
    Method applyDiscount(total)
        Return total / 2

Class MenuItem
    Attributes: name, price, TaxCalculator tax
    Method getPriceWithTax()
        Return price + tax.calculateTax(price)

Class Bill
    Attributes: items: List<MenuItem>, DiscountPolicy discount
    Method calculateTotal()
        subtotal = 0
        For each item in items
            subtotal += item.getPriceWithTax()
        finalAmount = discount.applyDiscount(subtotal)
        Return finalAmount

Main
    // Scenario 1
    item1 = new MenuItem("Paneer Tikka", 200, new FoodTax())
    item2 = new MenuItem("Coke", 100, new BeverageTax())
    item3 = new MenuItem("Imported Cheese", 500, new ImportedTax())
    bill1 = new Bill([item1, item2, item3], new PercentageDiscount())
    Print "Final Bill: " + bill1.calculateTotal()

    // Scenario 2: New discount scheme
    item4 = new MenuItem("Pizza", 300, new FoodTax())
    item5 = new MenuItem("Burger", 150, new FoodTax())
    bill2 = new Bill([item4, item5], new BuyOneGetOneDiscount())
    Print "Final Bill: " + bill2.calculateTotal()

    // Scenario 3: New LuxuryTax introduced
    item6 = new MenuItem("Imported Luxury Wine", 2000, new LuxuryTax())
    bill3 = new Bill([item6], new NoDiscount())
    Print "Final Bill: " + bill3.calculateTotal()
